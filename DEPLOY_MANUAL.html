5. Despliegue en Staging (209.46.123.188)

Crear base de datos

mysql -u root -p
CREATE DATABASE helix_comisiones_test
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

CREATE USER 'comisiones_user'@'localhost'
  IDENTIFIED BY 'TU_PASSWORD_STAGING';

GRANT ALL PRIVILEGES ON helix_comisiones_test.*
  TO 'comisiones_user'@'localhost';

FLUSH PRIVILEGES;
EXIT;

Crear estructura de directorios

mkdir -p /var/www/comisiones/frontend /var/www/comisiones/backend

Subir archivos al servidor (desde máquina local)

# Frontend (build de staging)
scp -r frontend/dist/frontend/browser/* \
  root@209.46.123.188:/var/www/comisiones/frontend/

# Backend
scp -r backend/dist root@209.46.123.188:/var/www/comisiones/backend/
scp backend/package.json root@209.46.123.188:/var/www/comisiones/backend/
scp backend/package-lock.json root@209.46.123.188:/var/www/comisiones/backend/
scp -r backend/prisma root@209.46.123.188:/var/www/comisiones/backend/

# Variables de entorno (subir .env.test como .env.production)
scp backend/.env.test root@209.46.123.188:/var/www/comisiones/backend/.env.production

Instalar dependencias y migrar base de datos

ssh root@209.46.123.188
cd /var/www/comisiones/backend

npm ci --omit=dev
npx prisma generate
NODE_ENV=production npx prisma migrate deploy

# Verificar arranque manual (Ctrl+C al confirmar)
NODE_ENV=production node dist/main.js

Configurar PM2

cat > /var/www/comisiones/backend/ecosystem.config.cjs << 'EOF'
module.exports = {
  apps: [{
    name: 'comisiones-backend',
    script: 'dist/main.js',
    cwd: '/var/www/comisiones/backend',
    instances: 1,
    exec_mode: 'fork',
    env: { NODE_ENV: 'production' },
    error_file: '/var/log/pm2/comisiones-error.log',
    out_file:   '/var/log/pm2/comisiones-out.log',
    log_date_format: 'YYYY-MM-DD HH:mm:ss'
  }]
}
EOF

mkdir -p /var/log/pm2
pm2 start /var/www/comisiones/backend/ecosystem.config.cjs
pm2 save
pm2 list

Configurar Apache

# Habilitar módulos necesarios
a2enmod proxy proxy_http headers rewrite

cat > /etc/apache2/sites-available/comisiones.staging.conf << 'EOF'
<VirtualHost *:80>
    ServerName staging.comisiones.alliantfed.us
    Redirect permanent / https://staging.comisiones.alliantfed.us/
</VirtualHost>

<VirtualHost *:443>
    ServerName staging.comisiones.alliantfed.us
    DocumentRoot /var/www/comisiones/frontend

    SSLEngine on
    SSLCertificateFile /etc/letsencrypt/live/staging.comisiones.alliantfed.us/fullchain.pem
    SSLCertificateKeyFile /etc/letsencrypt/live/staging.comisiones.alliantfed.us/privkey.pem

    ProxyPreserveHost On
    ProxyPass /api/v1 http://localhost:3001/api/v1
    ProxyPassReverse /api/v1 http://localhost:3001/api/v1

    <Directory /var/www/comisiones/frontend>
        Options -Indexes
        AllowOverride All
        Require all granted
        FallbackResource /index.html
    </Directory>

    Header always set X-Frame-Options "DENY"
    Header always set X-Content-Type-Options "nosniff"
</VirtualHost>
EOF

Obtener certificado SSL (staging — DNS only, certbot directo)

apt install certbot python3-certbot-apache -y

certbot --apache -d staging.comisiones.alliantfed.us \
  --non-interactive --agree-tos \
  --email tito@alliantfed.us

Activar sitio y verificar

a2ensite comisiones.staging.conf
apache2ctl configtest       # debe decir "Syntax OK"
systemctl reload apache2

# Verificar backend
curl -s -o /dev/null -w "%{http_code}" \
  https://staging.comisiones.alliantfed.us/api/v1/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"x","password":"x"}'
# Esperado: 401

SSL en producción — Certbot con DNS Challenge de Cloudflare

Como el dominio está proxied por Cloudflare, el challenge HTTP no llega al servidor. Se usa DNS challenge:

pip install certbot-dns-cloudflare

mkdir -p /root/.secrets/certbot
cat > /root/.secrets/certbot/cloudflare.ini << 'EOF'
dns_cloudflare_api_token = TU_CLOUDFLARE_API_TOKEN
EOF
chmod 600 /root/.secrets/certbot/cloudflare.ini

certbot certonly \
  --dns-cloudflare \
  --dns-cloudflare-credentials /root/.secrets/certbot/cloudflare.ini \
  -d comisiones.alliantfed.us \
  --non-interactive --agree-tos \
  --email tito@alliantfed.us
El API Token de Cloudflare necesita permiso Zone → DNS → Edit para alliantfed.us. Crearlo en Cloudflare → My Profile → API Tokens → Create Token.

7. Actualizaciones Futuras (Re-deploy)

# En el servidor de destino
pm2 stop comisiones-backend

# Reemplazar backend (dist/)
rm -rf /var/www/comisiones/backend/dist
# subir nuevo dist/ por SCP

# Si hay nuevas migraciones de base de datos
cd /var/www/comisiones/backend
NODE_ENV=production npx prisma migrate deploy

# Reiniciar proceso
pm2 restart comisiones-backend
pm2 logs comisiones-backend --lines 30

# Si hay cambios de frontend
rm -rf /var/www/comisiones/frontend/*
# subir nuevo build por SCP — Apache lo sirve sin restart
